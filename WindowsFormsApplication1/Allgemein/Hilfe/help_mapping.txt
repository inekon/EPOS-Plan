# ===========================================================================
# help_mapping.txt - Zuordnung Steuerelement -> Hilfeseite im Wiki
# ===========================================================================
#
# Diese Datei ist die EINZIGE Stelle, an der Hilfe gepflegt wird. Sie wird als
# eingebettete Ressource mitgeliefert (F2). Eine gleichnamige Datei NEBEN DER
# EXE uebersteuert die eingebettete Fassung vollstaendig - damit lassen sich
# Zuordnungen ohne Neubau korrigieren.
#
# Seit H3 gilt das woertlich: ein neuer Infobutton braucht KEINE Zeile
# Programmtext mehr. HilfeAutomatik (Allgemein\Hilfe\HilfeAutomatik.cs) erfasst
# jedes geoeffnete Formular und jedes nachgeladene UserControl von selbst (F5);
# es genuegt, hier eine Zeile zu ergaenzen. Der Infobutton muss dafuer nur mit
# "btn_Help" beginnen.
#
# Kodierung: UTF-8 (mit BOM). Beim Bearbeiten UTF-8 beibehalten - seit H2
# stehen Umlaute nicht mehr nur in den Kommentaren, sondern auch in den ZIELEN
# ("Waermebedarf" heisst als Kurzname "Wärmebedarf").
#
# ---------------------------------------------------------------------------
# Format (H2, Konzept Hilfesystem/Wikidokumentation, A3)
# ---------------------------------------------------------------------------
#
#   Praefix.Controlpfad = Kurzname[#anker]
#
#   Praefix       Name des Formulars (Form.Name) bzw. der Name, mit dem
#                 RegisterControl(container, name) aufgerufen wird.
#   Controlpfad   Name des Steuerelements. Zwischencontainer duerfen mit
#                 Punkten angegeben werden (ucPanel.btn_Help); die Suche
#                 laeuft ohnehin rekursiv ueber alle Ebenen.
#   Kurzname      Der Titelteil hinter "Programm Dokumentation/" der Wiki-
#                 Unterseite, die der Button oeffnen soll - also z. B.
#                 "Klimadaten" fuer
#                 https://wiki.epos-plan.de/wiki/Programm_Dokumentation/Klimadaten
#   #anker        Optional: eine Sprungmarke innerhalb der Seite
#                 ({{Anker|...}} im Wiki). Sie wird VOR der Katalogaufloesung
#                 abgetrennt und beim Oeffnen wieder angehaengt.
#
# ---------------------------------------------------------------------------
# Was sich mit H1/H2 geaendert hat
# ---------------------------------------------------------------------------
#
# 1. EIN Ziel je Zeile. Die frueher zweite Haelfte "| en-ziel" entfaellt: Es
#    werden keine englischen Wiki-Seiten gepflegt. Bei englischer Oberflaeche
#    leitet die Anwendung dieselbe Seite beim Oeffnen durch den
#    Uebersetzungs-Proxy (Entscheid 7.1a, Allgemein\Hilfe\DokuUebersetzung.cs).
#    Der Parser versteht die Paarform weiterhin - eine aeltere Datei neben der
#    EXE laeuft also nicht auf einen Fehler.
#
# 2. Kein WordPress mehr. Slugs und Link-Pfade von epos-plan.de sind ersatzlos
#    verworfen; es wurde nichts konvertiert. Der Katalog kennt ausschliesslich
#    die Unterseiten der Rubrik "Programm Dokumentation" - er laedt sie ueber
#    api.php?action=query&list=allpages&apprefix=Programm%20Dokumentation%2F.
#
# 3. Abgleich ueber die Kleinschreibform. Katalog-Kurzname und Mapping-Ziel
#    laufen beide durch WikiHelpCatalog.SlugNormalisieren (Trim +
#    ToLowerInvariant) - "Wärmebedarf", "wärmebedarf" und " Wärmebedarf "
#    treffen dieselbe Seite. Umlaute bleiben dabei Umlaute.
#
# 4. Zeilen ab '#' AM ZEILENANFANG sind Kommentar (nach Abzug von BOM,
#    Leerzeichen und Tabulatoren), Leerzeilen werden uebersprungen. Ein '#'
#    MITTEN in einer Zeile ist seit H2 die Sprungmarke und kein Kommentar.
#
# ---------------------------------------------------------------------------
# Vertragsliste
# ---------------------------------------------------------------------------
# 26 Infobuttons zeigen auf 23 Unterseiten. Das Gegenstueck steht im Wiki auf
# der Rubrikseite "Programm Dokumentation", Abschnitt "Hilfeseiten" - wer eine
# Seite umbenennt, sieht dort, was daran haengt. Umbenennungen im Wiki nur MIT
# bleibender Weiterleitung: ein ausgeliefertes Programm traegt diese Datei
# eingebettet bei sich und altert bis zum naechsten Release.
#
# ---------------------------------------------------------------------------
# Zuordnungen
# ---------------------------------------------------------------------------

# --- Startformular (Assistent) ------------------------------------------
# btn_Help sitzt NICHT im Klimaregion-Panel, sondern links davon direkt auf
# dem Formular, oberhalb des gesamten Reiterwerks (Projekt / Waermebedarf /
# Strombedarf / ...). Es ist die Hilfe zum Formular als Ganzem, also zum
# Ablauf des Programms - nicht zur Klimaregion (dafuer Form_Klimadaten) und
# nicht zur Kurzanleitung (dafuer btn_Help_Kurzanleitung auf Reiter 1).
Form_Start.btn_Help               = Programmablauf

# Reiter "Projekt": Einstieg in die Bedienung.
Form_Start.btn_Help_Kurzanleitung = Kurzanleitung

# Reiter "Waermebedarf".
Form_Start.btn_Help_Waermebedarf  = Wärmebedarf

# Reiter "Strombedarf".
Form_Start.btn_Help_Strombedarf   = Strombedarf

# --- Klimadaten ----------------------------------------------------------
Form_Klimadaten.btn_Help          = Klimadaten

# --- Kosten --------------------------------------------------------------
Form_Kosten.btn_Help              = Kosten

# ===========================================================================

# Simulation, dann der Rest. Je Bereich ist das fachliche Haupteinstiegs-
# fenster verdrahtet, nicht jedes Fenster des Bereichs.
#
# Anders als vor H2 gibt es keine "Notbehelfe" mehr: Jeder Dialog hat seine
# EIGENE Unterseite in der Rubrik (Entscheid 7.2). Wo mehrere Buttons auf
# dieselbe Seite zeigen (Waermebedarf, Projektverwaltung, Kurzanleitung), ist
# das eine bewusste Zusammenfassung und kein Ersatzziel.
